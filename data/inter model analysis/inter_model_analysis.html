agreement_summary <-map_dfr(all_features,function(f){
  w <-wide %>%
    select(case_id,model_short,value=all_of(f)) %>%
    pivot_wider(names_from=model_short,values_from=value) %>%
    drop_na()
  n_cases <-nrow(w)
  agree <-apply(w[,-1],1,function(r) length(unique(r))==1)
  tibble(feature=f,n_cases=n_cases,n_agree=sum(agree),
         n_disagree=n_cases-sum(agree),pct_agree=round(100*sum(agree)/n_cases,1))
}) %>% arrange(pct_agree)

High-disagreement cases

disagree_by_case <-map_dfr(active_features,function(f){
  w <-wide %>%
    select(case_id,model_short,value=all_of(f)) %>%
    pivot_wider(names_from=model_short,values_from=value) %>%
    drop_na()
  model_cols <-setdiff(names(w),"case_id")
  w %>% transmute(
    case_id,
    feature=f,
    disagree=apply(w[,model_cols],1,function(r) length(unique(r))>1)
  )
}) %>%
  group_by(case_id) %>%
  summarise(disagreement_score=sum(disagree),
            disagreed_features=paste(sort(feature[disagree]),collapse="; "),
            .groups="drop") %>%
  arrange(desc(disagreement_score))

high_disagreement <-disagree_by_case %>% filter(disagreement_score>=3)

Krippendorff’s alpha and pairwise Cohen’s kappa

kripp_by_feature <-map_dfr(all_features,function(f){
  m <-wide %>%
    select(case_id,model_short,value=all_of(f)) %>%
    pivot_wider(names_from=model_short,values_from=value) %>%
    drop_na() %>% select(-case_id) %>% as.matrix() %>% t()
  a <-tryCatch(kripp.alpha(m,method="nominal")$value,error=function(e) NA_real_)
  tibble(feature=f,krippendorff_alpha=round(a,3))
})

big <-map_dfr(active_features,function(f){
  wide %>% select(case_id,model_short,value=all_of(f)) %>% mutate(feature=f)

  unite("item",case_id,feature) %>%
  pivot_wider(names_from=model_short,values_from=value) %>%
  drop_na() %>% select(-item) %>% as.matrix() %>% t()
overall_alpha <-round(kripp.alpha(big,method="nominal")$value,3)

model_pairs <-list(
  c("GPT-5.4","Claude Opus 4.6"),
  c("GPT-5.4","Claude Sonnet 4.6"),
  c("Claude Opus 4.6","Claude Sonnet 4.6")
)
pairwise_kappa <-map_dfr(model_pairs,function(p){
  ks <-map_dbl(active_features,function(f){
    w <-wide %>% filter(model_short %in% p) %>%
      select(case_id,model_short,value=all_of(f)) %>%
      pivot_wider(names_from=model_short,values_from=value) %>% drop_na()
    tryCatch(kappa2(w[,p])$value,error=function(e) NA_real_)
  })
  tibble(pair=paste(p,collapse=" vs "),mean_kappa=round(mean(ks,na.rm=TRUE),3))
})

Bootstrap CI for overall alpha

set.seed(42)
boot_alpha <-replicate(2000,{
  idx <-sample(ncol(big),ncol(big),replace=TRUE)
  tryCatch(kripp.alpha(big[,idx],method="nominal")$value,error=function(e) NA_real_)
})
alpha_ci <-tibble(
  overall_alpha=overall_alpha,
  ci_low=round(quantile(boot_alpha,0.025,na.rm=TRUE),3),

SGLT2i agreement stratified by heart failure and CKD

strat_agree <-function(flag_col){
  map_dfr(c("0","1"),function(lvl){
    ids <-wide %>% filter(as.integer(str_to_lower(as.character(.data[[flag_col]]))=="yes")==as.integer(lvl)) %>% pull(case_id)
    w <-wide %>% filter(case_id %in% ids) %>%
      select(case_id,model_short,value=sglt2i_active) %>%
      pivot_wider(names_from=model_short,values_from=value) %>% drop_na()
    tibble(stratum=flag_col,present=lvl,n_cases=nrow(w),
           pct_agree=round(100*mean(apply(w[,-1],1,function(r) length(unique(r))==1)),1))
  })
}
sglt2_stratified <-bind_rows(strat_agree("heart_failure_present"),
                             strat_agree("chronic_kidney_disease"))

Disagreement vs clinical complexity

yesno <-function(x) as.integer(str_to_lower(as.character(x))=="yes")

case_clin <-wide %>%
  distinct(case_id,.keep_all=TRUE) %>%
  mutate(
    egfr_val=as.numeric(egfr_ml_min_1_73m2),
    comorbidity_count=
      yesno(heart_failure_present)+yesno(chronic_kidney_disease)+
      yesno(coronary_artery_disease)+yesno(mi_history)+
      yesno(stroke_tia_history)+yesno(cancer_history)
  ) %>%
  select(case_id,egfr_val,comorbidity_count) %>%
  left_join(disagree_by_case %>% select(case_id,disagreement_score),by="case_id")

cor_comorbid <-cor.test(case_clin$disagreement_score,case_clin$comorbidity_count,method="spearman")
cor_egfr     <-cor.test(case_clin$disagreement_score,case_clin$egfr_val,method="spearman")
